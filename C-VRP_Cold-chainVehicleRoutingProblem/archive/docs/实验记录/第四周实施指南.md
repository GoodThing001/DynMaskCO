## Day 1-2: 边影响分析

### 核心思路

**反事实实验（Counterfactual Analysis）**：
1. 对最终路线的每条边 `(u, v)`，暂时移除它
2. 用备用方法重建缺失部分
3. 比较移除前后的成本变化：`Δcost = cost_after - cost_original`
4. `Δcost` 越大 → 边越关键

### 边分类

| 类别 | 定义 | 含义 |
|------|------|------|
| **Critical** | Δcost > 10% | 移除后成本大幅上升，不可替代 |
| **Beneficial** | 0 < Δcost ≤ 10% | 有正向贡献 |
| **Neutral** | \|Δcost\| ≤ 0.1% | 可替代，影响很小 |
| **Harmful** | Δcost < 0 | ❌ 移除后反而更好（模型错误） |

### 实现代码

```python
"""
Week 4 v6: Edge Influence Analysis
边影响分析 - 通过反事实实验量化边的重要性
"""

import numpy as np
import matplotlib.pyplot as plt
from typing import List, Dict, Tuple
import pickle

def edge_influence_analysis(
    routes: np.ndarray,  # (batch, max_len) 路线
    dist_mat: np.ndarray,  # (batch, N, N) 距离矩阵
    demands: np.ndarray,  # (batch, N) 需求
    capacity: float,
    rebuild_method: str = 'greedy'  # 'greedy' | 'nearest'
) -> Dict:
    """
    对每条路线的每条边进行反事实分析
    
    Returns:
        influences: {
            batch_idx: {
                (u, v): {
                    'delta_cost': float,
                    'delta_pct': float,
                    'category': str,
                    'rebuilt_route': list
                }
            }
        }
    """
    batch_size = routes.shape[0]
    influences = {}
    
    for b_idx in range(batch_size):
        route = routes[b_idx]
        route = route[route > 0]  # 去除 padding
        
        # 提取边列表（包括返回 depot）
        edges = []
        prev = 0  # depot
        for node in route:
            if node == 0:  # 新车辆
                edges.append((prev, 0))  # 返回 depot
                prev = 0
            else:
                edges.append((prev, int(node)))
                prev = int(node)
        if prev != 0:
            edges.append((prev, 0))  # 最后返回 depot
        
        # 原始成本
        original_cost = evaluate_route_cost(route, dist_mat[b_idx])
        
        influences[b_idx] = {}
        
        for edge in edges:
            # 移除边，重建路线
            partial_route = remove_edge(route, edge)
            rebuilt_route = rebuild_route(
                partial_route, edge, dist_mat[b_idx], 
                demands[b_idx], capacity, method=rebuild_method
            )
            
            # 新成本
            new_cost = evaluate_route_cost(rebuilt_route, dist_mat[b_idx])
            delta = new_cost - original_cost
            delta_pct = (delta / original_cost * 100) if original_cost > 0 else 0
            
            # 分类
            if delta_pct > 10:
                category = 'critical'
            elif delta_pct > 0.1:
                category = 'beneficial'
            elif delta_pct > -0.1:
                category = 'neutral'
            else:
                category = 'harmful'
            
            influences[b_idx][edge] = {
                'delta_cost': delta,
                'delta_pct': delta_pct,
                'category': category,
                'rebuilt_route': rebuilt_route.tolist()
            }
    
    return influences

def remove_edge(route: np.ndarray, edge: Tuple[int, int]) -> np.ndarray:
    """从路线中移除指定的边"""
    u, v = edge
    route_list = route.tolist()
    
    # 找到边 (u, v) 的位置
    for i in range(len(route_list) - 1):
        if route_list[i] == u and route_list[i+1] == v:
            # 移除边：保留 u，移除 v
            # 注意：这会留下一个"缺口"，需要重建
            return np.array(route_list[:i+1] + route_list[i+2:])
    
    return route

def rebuild_route(
    partial_route: np.ndarray,
    missing_edge: Tuple[int, int],
    dist_mat: np.ndarray,
    demands: np.ndarray,
    capacity: float,
    method: str = 'greedy'
) -> np.ndarray:
    """
    重建路线：填补缺失的节点
    
    Methods:
    - 'greedy': 贪心插入（最小成本增加）
    - 'nearest': 最近邻插入
    """
    u, v = missing_edge
    
    if method == 'greedy':
        # 在 partial_route 中找到最佳插入位置
        best_cost = float('inf')
        best_pos = 0
        
        for pos in range(len(partial_route)):
            # 尝试在 pos 后插入 v
            if pos < len(partial_route) - 1:
                insert_cost = (dist_mat[partial_route[pos], v] +
                             dist_mat[v, partial_route[pos+1]] -
                             dist_mat[partial_route[pos], partial_route[pos+1]])
            else:
                insert_cost = dist_mat[partial_route[pos], v]
            
            if insert_cost < best_cost:
                best_cost = insert_cost
                best_pos = pos + 1
        
        # 插入
        rebuilt = np.insert(partial_route, best_pos, v)
        return rebuilt
    
    elif method == 'nearest':
        # 找到 partial_route 中离 v 最近的节点
        min_dist = float('inf')
        best_pos = 0
        
        for pos, node in enumerate(partial_route):
            if dist_mat[node, v] < min_dist:
                min_dist = dist_mat[node, v]
                best_pos = pos + 1
        
        rebuilt = np.insert(partial_route, best_pos, v)
        return rebuilt
    
    else:
        raise ValueError(f"Unknown rebuild method: {method}")

def evaluate_route_cost(route: np.ndarray, dist_mat: np.ndarray) -> float:
    """计算路线总距离成本"""
    cost = 0.0
    prev = 0  # depot
    for node in route:
        if node == 0:
            cost += dist_mat[prev, 0]
            prev = 0
        else:
            cost += dist_mat[prev, int(node)]
            prev = int(node)
    if prev != 0:
        cost += dist_mat[prev, 0]  # 返回 depot
    return cost

def analyze_statistics(influences: Dict) -> Dict:
    """统计分析边影响分布"""
    all_deltas = []
    categories = {'critical': 0, 'beneficial': 0, 'neutral': 0, 'harmful': 0}
    
    for batch_influences in influences.values():
        for edge_data in batch_influences.values():
            all_deltas.append(edge_data['delta_pct'])
            categories[edge_data['category']] += 1
    
    total = sum(categories.values())
    
    return {
        'mean_delta': np.mean(all_deltas),
        'std_delta': np.std(all_deltas),
        'median_delta': np.median(all_deltas),
        'min_delta': np.min(all_deltas),
        'max_delta': np.max(all_deltas),
        'categories': {k: v/total*100 for k, v in categories.items()},
        'total_edges': total
    }

def visualize_edge_influence(influences: Dict, output_path: str = None):
    """可视化边影响分布"""
    all_deltas = []
    for batch_influences in influences.values():
        for edge_data in batch_influences.values():
            all_deltas.append(edge_data['delta_pct'])
    
    plt.figure(figsize=(10, 6))
    
    # 直方图
    plt.hist(all_deltas, bins=50, edgecolor='black', alpha=0.7)
    plt.axvline(0, color='red', linestyle='--', linewidth=2, label='No change')
    plt.axvline(10, color='orange', linestyle='--', linewidth=1, label='Critical threshold')
    
    plt.xlabel('Cost Change (%)', fontsize=12)
    plt.ylabel('Number of Edges', fontsize=12)
    plt.title('Edge Influence Distribution (Counterfactual Analysis)', fontsize=14)
    plt.legend()
    plt.grid(alpha=0.3)
    
    if output_path:
        plt.savefig(output_path, dpi=300, bbox_inches='tight')
    plt.show()

# 使用示例
if __name__ == '__main__':
    # 加载测试数据
    data = np.load('C-VRP_Cold-chainVehicleRoutingProblem/data/p0_fix/dcc_50_r1_edod05_test.npz')
    
    # 加载已保存的解（来自 baseline 实验）
    # 假设我们保存了最佳解
    routes = data['routes'][:5]  # 前 5 个实例
    dist_mat = data['dist_mat'][:5]
    demands = data['demands'][:5]
    capacity = 50.0
    
    # 执行边影响分析
    print("Running edge influence analysis...")
    influences = edge_influence_analysis(
        routes, dist_mat, demands, capacity, rebuild_method='greedy'
    )
    
    # 统计分析
    stats = analyze_statistics(influences)
    print("\n=== Edge Influence Statistics ===")
    print(f"Total edges analyzed: {stats['total_edges']}")
    print(f"Mean Δcost: {stats['mean_delta']:.2f}%")
    print(f"Std Δcost: {stats['std_delta']:.2f}%")
    print(f"Median Δcost: {stats['median_delta']:.2f}%")
    print(f"Range: [{stats['min_delta']:.2f}%, {stats['max_delta']:.2f}%]")
    print(f"\nCategory Distribution:")
    for cat, pct in stats['categories'].items():
        print(f"  {cat.capitalize()}: {pct:.1f}%")
    
    # 可视化
    visualize_edge_influence(influences, output_path='edge_influence_dist.png')
    
    # 保存结果
    with open('edge_influences.pkl', 'wb') as f:
        pickle.dump(influences, f)
    print("\nResults saved to edge_influences.pkl")
```

#### 实验 1：温度类别置换

**操作**：交换两个客户的温度类别（冷冻 ↔ 常温）

```python
def temperature_swap_experiment(data, model, instance_ids=[0, 1, 2]):
    """
    温度类别置换实验
    
    预期：路线结构应该显著改变（overlap < 80%）
    """
    results = []
    
    for inst_id in instance_ids:
        # 原始数据
        original_data = extract_instance(data, inst_id)
        original_solution = model.solve(original_data)
        
        # 扰动：随机选 2 个节点交换温度类别
        perturbed_data = original_data.copy()
        nodes = np.random.choice(range(1, 51), size=2, replace=False)
        perturbed_data['temp_class'][nodes[0]], perturbed_data['temp_class'][nodes[1]] = \
            perturbed_data['temp_class'][nodes[1]], perturbed_data['temp_class'][nodes[0]]
        
        # 重新求解
        perturbed_solution = model.solve(perturbed_data)
        
        # 分析
        overlap = compute_route_overlap(original_solution, perturbed_solution)
        cost_change = (perturbed_solution['cost'] - original_solution['cost']) / original_solution['cost'] * 100
        
        results.append({
            'instance': inst_id,
            'swapped_nodes': nodes.tolist(),
            'overlap': overlap,
            'cost_change_pct': cost_change,
            'original_temp': original_data['temp_class'][nodes].tolist(),
            'perturbed_temp': perturbed_data['temp_class'][nodes].tolist()
        })
    
    return results

def compute_route_overlap(sol1, sol2):
    """计算两条路线的重叠度（边的交集）"""
    edges1 = set(extract_edges(sol1['route']))
    edges2 = set(extract_edges(sol2['route']))
    
    intersection = len(edges1 & edges2)
    union = len(edges1 | edges2)
    
    return intersection / union if union > 0 else 0
```

#### 实验 2：时间窗宽松化

```python
def tw_relaxation_experiment(data, model, instance_ids=[0, 1, 2]):
    """
    时间窗宽松化实验
    
    预期：
    - 可行性提高 (违例率下降)
    - 成本降低 (有更多优化空间)
    """
    results = []
    
    for inst_id in instance_ids:
        # 原始数据
        original_data = extract_instance(data, inst_id)
        original_solution = model.solve(original_data)
        
        # 扰动：扩大 TW 宽度 ×2
        perturbed_data = original_data.copy()
        mid = (perturbed_data['tw_start'] + perturbed_data['tw_end']) / 2
        width = perturbed_data['tw_end'] - perturbed_data['tw_start']
        perturbed_data['tw_start'] = mid - width  # 扩大一倍
        perturbed_data['tw_end'] = mid + width
        
        # 重新求解
        perturbed_solution = model.solve(perturbed_data)
        
        # 分析
        feas_change = perturbed_solution['feasible'] - original_solution['feasible']
        cost_change = (perturbed_solution['cost'] - original_solution['cost']) / original_solution['cost'] * 100
        
        results.append({
            'instance': inst_id,
            'original_feas': original_solution['feasible'],
            'perturbed_feas': perturbed_solution['feasible'],
            'feas_change': feas_change,
            'cost_change_pct': cost_change,
            'tw_width_multiplier': 2.0
        })
    
    return results
```
